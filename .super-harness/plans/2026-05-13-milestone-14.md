# capability/command/builtin/voice.py
"""/voice 命令 — 开启/关闭语音交互模式。"""

from __future__ import annotations

from typing import Any

from capability.command.base import (
    Command,
    CommandContext,
    CommandMeta,
    CommandResult,
    CommandResultType,
    CommandType,
)


class VoiceCommand(Command):
    """/voice 命令 — 控制语音交互模式。"""

    def __init__(self, voice_runtime: Any = None) -> None:
        self._voice_runtime = voice_runtime

    @property
    def meta(self) -> CommandMeta:
        return CommandMeta(
            name="voice",
            description="开启/关闭语音交互模式 (voice input/output)",
            type=CommandType.SYSTEM,
        )

    async def execute(self, args: str, context: CommandContext) -> CommandResult:
        if self._voice_runtime is None:
            context.console.print("[yellow]语音模块未启用。请在配置中设置 voice.enabled = true[/]")
            return CommandResult(type=CommandResultType.NONE)

        # 解析参数
        continuous = "--continuous" in args

        if self._voice_runtime.is_running:
            await self._voice_runtime.stop()
            context.console.print("[dim]语音模式已关闭[/]")
        else:
            await self._voice_runtime.start(continuous=continuous)
            mode = "连续对话" if continuous else "唤醒模式"
            context.console.print(f"[green]语音模式已开启 ({mode})[/]")

        return CommandResult(type=CommandResultType.NONE)
```

# 替换原来的 session.prompt_async() 为竞争模式
while True:
    if not first_turn:
        self._console.print()
    first_turn = False

    # 并行等待键盘输入和语音输入
    keyboard_task = asyncio.create_task(
        session.prompt_async(HTML('<ansicyan><b>❯ </b></ansicyan>'))
    )
    voice_task = asyncio.create_task(self._voice_input_queue.get())

    try:
        done, pending = await asyncio.wait(
            [keyboard_task, voice_task],
            return_when=asyncio.FIRST_COMPLETED,
        )
    except Exception:
        keyboard_task.cancel()
        voice_task.cancel()
        raise

    for task in pending:
        task.cancel()
        try:
            await task
        except (asyncio.CancelledError, Exception):
            pass

    if voice_task in done:
        text = voice_task.result()
        self._console.print(f"[dim]🎤 {text}[/]")
    elif keyboard_task in done:
        try:
            text = keyboard_task.result()
        except (EOFError, KeyboardInterrupt):
            self._memory.save_session()
            self._console.print("\n[dim]Goodbye.[/]")
            break
    else:
        continue

    text = text.strip()
    if not text:
        continue

    # Slash commands
    if text.startswith("/"):
        if await self._handle_command(text):
            continue

    # Normal message
    self._console.print()
    try:
        await self._process_streaming(text)
    except Exception as exc:
        self._console.print("[red]Error:[/] {}".format(exc))
```

class VoiceRuntimeImpl:
    # ... 现有代码 ...

    def __init__(self, config, hooks, llm_fn):
        # ... 现有初始化 ...
        self._continuous = False
        self._idle_timeout_task: asyncio.Task | None = None

    async def start(self, continuous: bool = False) -> None:
        """启动语音 Pipeline。"""
        self._continuous = continuous
        # ... 现有启动逻辑 ...

    def _on_tts_complete(self) -> None:
        """TTS 播放完成回调。"""
        if self._continuous:
            self._state = VoiceConversationState.SLEEPING
            self._hooks.fire("voice_state", state="sleeping")
            # 启动空闲超时检测
            self._idle_timeout_task = asyncio.create_task(
                self._idle_timeout_loop()
            )
        else:
            self._state = VoiceConversationState.IDLE
            self._hooks.fire("voice_state", state="idle")

    async def _idle_timeout_loop(self) -> None:
        """空闲超时检测 — 连续模式下无语音则回到 IDLE。"""
        timeout = self._config.continuous_idle_timeout
        try:
            await asyncio.sleep(timeout)
            # 超时，回到 IDLE
            self._state = VoiceConversationState.IDLE
            self._hooks.fire("voice_state", state="idle")
            if self._on_state_change:
                self._on_state_change("idle")
        except asyncio.CancelledError:
            pass  # 被取消说明有新的语音输入

    def _handle_voice_detected(self) -> None:
        """检测到语音输入时取消超时。"""
        if self._idle_timeout_task is not None:
            self._idle_timeout_task.cancel()
            self._idle_timeout_task = None
        self._state = VoiceConversationState.LISTENING
